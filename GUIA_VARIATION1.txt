OPCIÓN 3️⃣: Uso Programático (Para desarrolladores)
────────────────────────────────────────────────────────────────────────────────

from src.calculadora_v2 import CalculadoraCETESV2

# Crear calculadora CON ingreso mensual
calc = CalculadoraCETESV2(
    capital=250000,
    tasa_cetes=0.0719,
    tasa_inflacion=0.0411,
    ingreso_bruto_mensual=35000  # ← Sistema calcula ISR automáticamente
)

# Calcular
resultado = calc.calcular()

# Ver resultado detallado
print(calc.resumen_texto_detallado())

# Obtener solo la información de ISR
info_isr = calc.obtener_informacion_isr()
print(f"Tasa ISR: {info_isr['porcentaje_excedente']*100:.2f}%")
